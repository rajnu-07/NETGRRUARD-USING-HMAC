Flask Middleware Code
from flask import Flask, request import time, redis, sqlite3 app = Flask(__name__) r = redis.Redis() @app.before_request # fires for EVERY route automatically def log_request(): entry = { "ip": request.remote_addr, "method": request.method, "path": request.path, "timestamp": time.time(), "user_agent":request.headers.get("User-Agent"), "payload_sz":request.content_length or 0 } key = f"rate:{entry['ip']}" r.incr(key) # atomic increment r.expire(key, 60) # auto-reset after 60s write_sqlite(entry) # permanent record
